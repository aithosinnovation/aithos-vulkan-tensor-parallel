cmake_minimum_required(VERSION 3.19)
project(aithos_correctness LANGUAGES CXX)
add_executable(aithos-controls controls.cpp)
target_compile_features(aithos-controls PRIVATE cxx_std_17)
target_compile_definitions(aithos-controls PRIVATE LLAMA_SHARED GGML_SHARED)
target_include_directories(aithos-controls PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include")
find_library(AITHOS_LLAMA llama PATHS "${LLAMA_BUILD}/src" "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
find_library(AITHOS_GGML ggml PATHS "${LLAMA_BUILD}/ggml/src" "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
find_library(AITHOS_BASE ggml-base PATHS "${LLAMA_BUILD}/ggml/src" "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(aithos-controls PRIVATE "${AITHOS_LLAMA}" "${AITHOS_GGML}" "${AITHOS_BASE}")
set_target_properties(aithos-controls PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${LLAMA_BUILD}/bin")
