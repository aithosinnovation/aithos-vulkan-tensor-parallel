cmake_minimum_required(VERSION 3.19)
project(aithos_correctness LANGUAGES CXX)
add_executable(aithos-mixed-repeat mixed-repeat.cpp)
target_compile_features(aithos-mixed-repeat PRIVATE cxx_std_17)
target_compile_definitions(aithos-mixed-repeat PRIVATE LLAMA_SHARED GGML_SHARED)
target_include_directories(aithos-mixed-repeat PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include")
find_library(AITHOS_LLAMA llama PATHS "${LLAMA_BUILD}/src" "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
find_library(AITHOS_GGML ggml PATHS "${LLAMA_BUILD}/ggml/src" "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
find_library(AITHOS_BASE ggml-base PATHS "${LLAMA_BUILD}/ggml/src" "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(aithos-mixed-repeat PRIVATE "${AITHOS_LLAMA}" "${AITHOS_GGML}" "${AITHOS_BASE}")
set_target_properties(aithos-mixed-repeat PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${LLAMA_BUILD}/bin")
